function renderizarProductos(lista) { const grid = document.getElementById('productsGrid'); if (!lista || lista.length === 0) { grid.innerHTML = '
No se encontraron productos.
'; return; } grid.innerHTML = lista.map((p, index) => { // 1. Parsear imágenes (separadas por coma) let imgs = []; if (p.url_imagen && String(p.url_imagen).trim() !== '') { imgs = String(p.url_imagen).split(',').map(u => u.trim()).filter(u => u.length > 0); } if (imgs.length === 0) { imgs = ['https://via.placeholder.com/300?text=Colora']; } // Guardar el estado de la imagen actual para cada tarjeta const prodKey = p.id || index; imagenesEstado[prodKey] = { fotos: imgs, idx: 0 }; // 2. Precio, Stock y Presentación Comercial (unidad_medida, contenido_cant, contenido_unidad) const precio = parseFloat(String(p.precio || 0).replace(',', '.')) || 0; const stockNum = parseInt(p.stock) || 0; const sinStock = stockNum <= 0; let htmlPresentacion = ''; if (p.unidad_medida && p.contenido_cant) { const medida = String(p.unidad_medida).trim(); const cantidad = p.contenido_cant; const unidadContenido = p.contenido_unidad ? ` ${String(p.contenido_unidad).trim()}` : ''; htmlPresentacion = `
📦 ${medida}: ${cantidad}${unidadContenido}
`; } else if (p.unidad_medida) { htmlPresentacion = `
📦 ${String(p.unidad_medida).trim()}
`; } return `

${p.nombre || 'Sin nombre'}

${htmlPresentacion}

${p.descripcion || ''}

S/ ${precio.toFixed(2)}
`; }).join(''); }
